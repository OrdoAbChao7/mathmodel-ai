for h in range(24):
    terms = [(processing[h], 1)]
    for s in starts:
        if s <= h < s + 8:

# b_h = b_(h-1) + arrival_h - processing_h

    terms = [(backlog[h], 1), (processing[h], 1)]
    if h > 0:
        terms += [(backlog[h-1], -1)]
    add_constraint(terms, lower=arrival[h], upper=arrival[h])
